Implementation (adapted from PPTX2HTML `scripts/src/util/zip.js` approach; their 221-line hand-rolled inflate is replaced by `node:zlib`):

```js
import { deflateRawSync, inflateRawSync } from 'node:zlib';

const EOCD_SIG = 0x06054b50, CEN_SIG = 0x02014b50, LOC_SIG = 0x04034b50,
      ZIP64_EOCD_SIG = 0x06064b50, ZIP64_LOC_SIG = 0x07064b50;
const OLE2_MAGIC = Buffer.from([0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1]);

// Adapted from PPTX2HTML (MIT): standard CRC-32 table.
const CRC_TABLE = (() => {
  const t = new Int32Array(256);
  for (let n = 0; n < 256; n++) {
    let c = n;
    for (let k = 0; k < 8; k++) c = c & 1 ? (0xedb88320 ^ (c >>> 1)) : (c >>> 1);
    t[n] = c;
  }
  return t;
})();
export function crc32(buf, crc = 0) {
  crc = ~crc;
  for (let i = 0; i < buf.length; i++) crc = CRC_TABLE[(crc ^ buf[i]) & 0xff] ^ (crc >>> 8);
  return ~crc >>> 0;
}

export class ZipReader {
  constructor(data) {
    if (OLE2_MAGIC.equals(Buffer.from(data.buffer, data.byteOffset, 8)))
      throw new Error('This is a legacy .ppt (OLE2) file — re-save it as .pptx first');
    this.data = data instanceof Buffer ? data : Buffer.from(data);
    this.view = new DataView(this.data.buffer, this.data.byteOffset, this.data.byteLength);
    this.entries = new Map(); // name -> { name, method, csize, usize, off }
    this.#readDirectory();
  }
  has(name) { return this.entries.has(name); }
  names() { return [...this.entries.keys()]; }
  read(name) {
    const e = this.entries.get(name);
    if (!e) throw new Error(`zip: no entry ${name}`);
    const d = this.data;
    if (this.#u32(e.off) !== LOC_SIG) throw new Error(`zip: bad local header for ${name}`);
    const nameLen = this.#u16(e.off + 26), exLen = this.#u16(e.off + 28);
    const start = e.off + 30 + nameLen + exLen;
    const raw = d.subarray(start, start + e.csize);
    if (e.method === 0) return new Uint8Array(raw);
    if (e.method === 8) return new Uint8Array(inflateRawSync(raw));
    throw new Error(`zip: unsupported compression method ${e.method} for ${name}`);
  }
  #u16(o) { return this.view.getUint16(o, true); }
  #u32(o) { return this.view.getUint32(o, true); }
  #u64(o) { return Number(this.view.getBigUint64(o, true)); }
  #findEocd() { // scan backwards within the max 64 KiB + 22-byte comment window
    const d = this.data, min = Math.max(0, d.length - 65557);
    for (let i = d.length - 22; i >= min; i--) if (this.#u32(i) === EOCD_SIG) return i;
    return -1;
  }
  #readDirectory() {
    const eocd = this.#findEocd();
    if (eocd < 0) { this.#recoverLocalHeaders(); return; }
    let count = this.#u16(eocd + 10), size = this.#u32(eocd + 12), off = this.#u32(eocd + 16);
    if (count === 0xffff || size === 0xffffffff || off === 0xffffffff) {
      // ZIP64: locator sits immediately before EOCD
      const loc = eocd - 20;
      if (loc >= 0 && this.#u32(loc) === ZIP64_LOC_SIG) {
        const z64 = this.#u64(loc + 8);
        if (this.#u32(z64) === ZIP64_EOCD_SIG) {
          count = this.#u64(z64 + 32); size = this.#u64(z64 + 40); off = this.#u64(z64 + 48);
        }
      }
    }
    if (this.#u32(off) !== CEN_SIG) { this.#recoverLocalHeaders(); return; }
    let p = off;
    for (let i = 0; i < count && p + 46 <= this.data.length; i++) {
      if (this.#u32(p) !== CEN_SIG) break;
      const flags = this.#u16(p + 8), method = this.#u16(p + 10);
      let csize = this.#u32(p + 20), usize = this.#u32(p + 24);
      const nameLen = this.#u16(p + 28), exLen = this.#u16(p + 30), cmtLen = this.#u16(p + 32);
      let lho = this.#u32(p + 42);
      const name = this.data.toString(flags & 0x800 ? 'utf8' : 'latin1', p + 46, p + 46 + nameLen);
      // ZIP64 extra field 0x0001 overrides the 0xFFFFFFFF sentinels
      let q = p + 46 + nameLen; const end = q + exLen;
      while (q + 4 <= end) {
        const id = this.#u16(q), sz = this.#u16(q + 2);
        if (id === 1) {
          let r = q + 4;
          if (usize === 0xffffffff) { usize = this.#u64(r); r += 8; }
          if (csize === 0xffffffff) { csize = this.#u64(r); r += 8; }
          if (lho === 0xffffffff) lho = this.#u64(r);
        }
        q += 4 + sz;
      }
      if (flags & 0x1) throw new Error(`zip: entry ${name} is encrypted — unsupported`);
      this.entries.set(name, { name, method, csize, usize, off: lho });
      p += 46 + nameLen + exLen + cmtLen;
    }
  }
  #recoverLocalHeaders() { // truncated archive: walk local headers sequentially
    let p = 0;
    while (p + 30 <= this.data.length && this.#u32(p) === LOC_SIG) {
      const flags = this.#u16(p + 6), method = this.#u16(p + 8);
      const csize = this.#u32(p + 18), usize = this.#u32(p + 22);
      const nameLen = this.#u16(p + 26), exLen = this.#u16(p + 28);
      const name = this.data.toString(flags & 0x800 ? 'utf8' : 'latin1', p + 30, p + 30 + nameLen);
      this.entries.set(name, { name, method, csize, usize, off: p });
      p += 30 + nameLen + exLen + (csize || usize);
      if (!csize && !usize) break; // data descriptor without sizes: cannot skip safely
    }
    if (!this.entries.size) throw new Error('zip: no readable entries (not a ZIP/OOXML file)');
  }
}

export function writeZip(files) {
  const chunks = [], central = []; let off = 0;
  for (const { name, data } of files) {
    const nb = Buffer.from(name, 'utf8');
    const db = Buffer.isBuffer(data) ? data : Buffer.from(data);
    const comp = deflateRawSync(db), crc = crc32(db);
    const lh = Buffer.alloc(30);
    lh.writeUInt32LE(LOC_SIG, 0); lh.writeUInt16LE(20, 4); lh.writeUInt16LE(0x0800, 6);
    lh.writeUInt16LE(8, 8); lh.writeUInt32LE(crc, 14); lh.writeUInt32LE(comp.length, 18);
    lh.writeUInt32LE(db.length, 22); lh.writeUInt16LE(nb.length, 26); lh.writeUInt16LE(0, 28);
    chunks.push(lh, nb, comp);
    const ch = Buffer.alloc(46);
    ch.writeUInt32LE(CEN_SIG, 0); ch.writeUInt16LE(20, 4); ch.writeUInt16LE(20, 6);
    ch.writeUInt16LE(0x0800, 8); ch.writeUInt16LE(8, 10); ch.writeUInt32LE(crc, 16);
    ch.writeUInt32LE(comp.length, 20); ch.writeUInt32LE(db.length, 24);
    ch.writeUInt16LE(nb.length, 28); ch.writeUInt32LE(off, 42);
    central.push(Buffer.concat([ch, nb]));
    off += 30 + nb.length + comp.length;
  }
  const cd = Buffer.concat(central);
  const eocd = Buffer.alloc(22);
  eocd.writeUInt32LE(EOCD_SIG, 0); eocd.writeUInt16LE(files.length, 8);
  eocd.writeUInt16LE(files.length, 10); eocd.writeUInt32LE(cd.length, 12); eocd.writeUInt32LE(off, 16);
  return Buffer.concat([...chunks, cd, eocd]);
}
```

The design decision that makes the whole converter simple (adapted from PPTX2HTML `scripts/src/util/xml.js`, MIT): **canonicalize every element/attribute name to a fixed prefix set** so downstream code matches `'a:solidFill'` by string regardless of producer prefixes or transitional-vs-strict OOXML. Canonical URIs:

```js
// prefix -> namespace URI, both transitional and strict (ECMA-376) variants
const CANON = {
  a: ['http://schemas.openxmlformats.org/drawingml/2006/main', 'http://purl.oclc.org/ooxml/drawingml/main'],
  p: ['http://schemas.openxmlformats.org/presentationml/2006/main', 'http://purl.oclc.org/ooxml/presentationml/main'],
  r: ['http://schemas.openxmlformats.org/officeDocument/2006/relationships', 'http://purl.oclc.org/ooxml/officeDocument/relationships'],
  c: ['http://schemas.openxmlformats.org/drawingml/2006/chart'],
  cx: ['http://schemas.microsoft.com/office/drawing/2014/chartex'],
  dgm: ['http://schemas.openxmlformats.org/drawingml/2006/diagram'],
  dsp: ['http://schemas.microsoft.com/office/drawing/2008/diagram'],
  p14: ['http://schemas.microsoft.com/office/powerpoint/2010/main'],
  p15: ['http://schemas.microsoft.com/office/powerpoint/2012/main'],
  a14: ['http://schemas.microsoft.com/office/drawing/2010/main'],
  mc: ['http://schemas.openxmlformats.org/markup-compatibility/2006'],
};
const URI_TO_CANON = new Map(Object.entries(CANON).flatMap(([pfx, uris]) => uris.map(u => [u, pfx])));
```

Parser behavior:
1. Skip BOM, `<?xml ... ?>`, `<!-- ... -->` (no nesting), `<!DOCTYPE ...>`, CDATA becomes text content. No DTD/entity expansion — treat `<!ENTITY>` as unsupported and throw.
2. Tokenize tags with a scan loop (not regex-over-whole-doc): `<name attr="v" attr='v' attr=bare ...>` self-closing or paired; maintain a stack; text runs between tags append to the current node's text buffer. Decode entities in text AND attribute values: `&lt; &gt; &amp; &quot; &apos;` plus numeric `&#123;` / `&#x1A;` (Office writes odd numeric escapes — upstream calls this `decodeOoxmlEscapes`).
3. While parsing, track in-scope `xmlns` and `xmlns:PFX` declarations in a scope stack; resolve each element name and each *prefixed attribute* name to its URI. After the parse completes, walk the tree once and rewrite `node.name` to `(URI_TO_CANON.get(uri) ?? originalPrefix) + ':' + local`. Unprefixed element names with no default xmlns stay unprefixed. Attributes in no namespace keep their bare name; `xmlns*` attributes are dropped from `attrs`.
4. `XNode` shape and helpers — exact signatures:

```js
export class XNode {
  constructor(name) { this.name = name; this.attrs = {}; this.children = []; this.parent = null; this.text = ''; }
  all(name)          // direct children with canonical name `name`; accepts single name or array of names
  find(name)         // first `all(name)` child or null
  path(spec)         // '/'-separated canonical names, e.g. 'p:cSld/p:spTree'; first match or null
  textOf()           // concatenated direct text, entity-decoded
  attr(k, d = null)  // attrs[k] ?? d
  num(k, d = 0)      // parseFloat(attr) with fallback d
  int(k, d = 0)      // parseInt
  bool(k, d = false) // attr === '1' || attr === 'true'
}
export function escapeXml(s) // &, <, >, ", ' — used by every renderer later
```

```js
import { parseXml } from '../util/xml.js';
import { sniffMime, MIME_BY_EXT } from '../util/mime.js';

const OFFICE_DOCUMENT = 'http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument';

export class OpcPackage {
  /** @param {Uint8Array} bytes of the .pptx */
  static open(bytes) { return new OpcPackage(bytes); }
  constructor(bytes) {
    this.zip = new ZipReader(bytes);          // from task 1
    this.ctDefaults = {};                     // ext -> mime
    this.ctOverrides = {};                    // partname -> mime
    this.xmlCache = new Map();                // partname -> XNode
    this.relsCache = new Map();               // rels partname -> [{ id, type, target, external }]
    if (!this.zip.has('[Content_Types].xml'))
      throw new Error('pptx: missing [Content_Types].xml — not an OOXML package');
    for (const d of this.xml('[Content_Types].xml').find('Types').all(['Default','Override']))
      if (d.name === 'Default') this.ctDefaults[d.attr('Extension').toLowerCase()] = d.attr('ContentType');
      else this.ctOverrides[normalize(d.attr('PartName'))] = d.attr('ContentType');
  }
  has(name) { return this.zip.has(name); }
  bytes(name) { /* zip.read(name), throws Error(`pptx: missing part ${name}`) if !has */ }
  text(name) { Buffer.from(this.bytes(name)).toString('utf8') }
  xml(name)  { cached parseXml(this.text(name)) — one XNode per part, reused }
  rels(partName) // map '<dir>/_rels/<base>.rels' from partName; cached; returns [] if the .rels part is absent
  relBy(typeSuffixOrFull, partName) // first rel whose Type ends with the suffix, e.g. relBy('slideLayout', 'ppt/slides/slide1.xml')
  resolvePartName(fromPart, target) // see below
  mimeOf(partName)  // sniffMime(bytes) ?? (override ?? default-by-ext ?? MIME_BY_EXT ?? 'application/octet-stream') — lazily reads bytes only for binary parts (has a non-xml override mime); never throws
  mainPart()        // package rels ('_rels/.rels') rel Type === OFFICE_DOCUMENT → resolve → normally '/ppt/presentation.xml'
}
```

`resolvePartName(fromPart, target)` — the upstream algorithm: if `target` starts with `/` return normalized absolute; else resolve relative to `fromPart`'s directory, splitting on `/`, consuming `..` and `.` segments; percent-decode each segment (`decodeURIComponent` in try/catch — a literal `%` must not crash the converter); strip leading `./`. Return the normalized part name *without* leading slash (zip entry names have none). `rels()` must skip entries with `TargetMode="External"` (keep them in the list with `external: true` so hyperlink code can use them, but `resolvePartName` is never called for them).

**Tolerance contract (round-trip critical, research Q2):** `[Content_Types].xml` may advertise overrides for parts that do not exist (pptxgenjs quirk). The package itself must never assume an advertised part exists: `ctOverrides` is just a map; every consumer calls `has()` before `bytes()/xml()`. Encode this as a test.

`normalize(partName)` = `partName.replace(/^\//, '')`.
