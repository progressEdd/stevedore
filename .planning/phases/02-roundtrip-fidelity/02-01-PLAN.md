**Step 1 — create `scripts/vendor-sync.sh`** (the durable mirror tool later plans re-run after every fork-branch fix):

```bash
#!/usr/bin/env bash
# scripts/vendor-sync.sh — mirror a fork checkout's tree into vendor/<name>.
# Usage: scripts/vendor-sync.sh <source-checkout-dir> <vendor-name>
# Excludes installs/build outputs and, for PPTX2HTML, the heavy samples/
# tree (pptx binaries — available in the fork on GitHub; keeps the
# subtree-distributed skill light). --delete mirror: upstream removals
# propagate. Commit the result on the stevedore branch.
set -euo pipefail
[ $# -eq 2 ] || { echo "usage: $0 <src-dir> <vendor-name>" >&2; exit 2; }
SRC="$(realpath "$1")"; NAME="$2"
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
DEST="$ROOT/vendor/$NAME"
EXCLUDES=(--exclude node_modules --exclude dist --exclude .git --exclude bun.lock --exclude bun.lockb)
[ "$NAME" = "PPTX2HTML" ] && EXCLUDES+=(--exclude samples)
mkdir -p "$DEST"
rsync -a --delete "${EXCLUDES[@]}" "$SRC"/ "$DEST"/
echo "SYNCED vendor/$NAME <- $SRC"
```

**Step 1 — `scripts/setup-roundtrip.sh`** (the "one setup script" of SC1; idempotent):

```bash
#!/usr/bin/env bash
# scripts/setup-roundtrip.sh — one-time prep of the combined repo's toolchains.
# Idempotent — safe to re-run any time.
#   1. vendor/dom-to-pptx (html→pptx half): its own puppeteer toolchain
#      (pinned on the fork branch — vendored tool's internals untouched,
#      decision 10) + build.
#   2. vendor/PPTX2HTML devDeps: playwright — the driver for OUR export
#      tooling (decision 13; same driver as PPTX2HTML's playwright+pngjs
#      deterministic-PNG verify pattern). Bundled browsers are NEVER
#      downloaded (NixOS): nixpkgs chromium is passed at run time via
#      executablePath (STEVEDORE_CHROMIUM_PATH, set by the nix-shell wrapper).
set -euo pipefail
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VENDOR="$ROOT/vendor/dom-to-pptx"
PPT="$ROOT/vendor/PPTX2HTML"
command -v bun >/dev/null || { echo "setup: bun not found (npm/npx are not installed on this machine)" >&2; exit 1; }
command -v nix >/dev/null || { echo "setup: nix not found (chromium comes from nixpkgs)" >&2; exit 1; }
nix shell nixpkgs#chromium --command chromium --version >/dev/null \
  || { echo "setup: chromium unavailable via 'nix shell nixpkgs#chromium'" >&2; exit 1; }
cd "$VENDOR"
PUPPETEER_SKIP_DOWNLOAD=1 bun install
bun run build   # the repo ships no dist/; the exporter CLI refuses to run without it
[ -n "$(ls -A dist 2>/dev/null)" ] || { echo "setup: bun run build produced no dist/ — inspect $VENDOR/dist" >&2; exit 1; }
cd "$PPT"
# devDeps (esbuild/playwright/pngjs) — playwright here is OUR exporter's driver
# (02-03, decision 13). The vendored package.json itself stays untouched.
PLAYWRIGHT_SKIP_BROWSER_DOWNLOAD=1 bun install
[ -d node_modules/playwright ] || { echo "setup: playwright did not install under $PPT — vendor/PPTX2HTML devDependencies do not match phase-01 research (expected esbuild/playwright/pngjs); STOP and report instead of editing the vendored package.json" >&2; exit 1; }
echo "SETUP-OK"
```

`chmod +x` it.

**Step 2 — `scripts/roundtrip.sh`** (stages 1–2 + compare now; plan 02-03 inserts the PNG-export stage before the compare):

```bash
#!/usr/bin/env bash
# scripts/roundtrip.sh — the whole round-trip from the combined repo (no /tmp
# tool cloning: tools are vendored; only outputs are scratch).
# Usage: scripts/roundtrip.sh <source.html> <outdir>
# Stages: html → pptx (vendor/dom-to-pptx, nix chromium) → html (our converter)
#         [→ per-slide PNG exports of both sides — added by plan 02-03]
#         → comparator (scripts/compare-decks.mjs).
# Run scripts/setup-roundtrip.sh once first.
set -euo pipefail
[ $# -eq 2 ] || { echo "usage: $0 <source.html> <outdir>" >&2; exit 2; }
SRC="$(realpath "$1")"; OUT="$(realpath -m "$2")"
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VENDOR="$ROOT/vendor/dom-to-pptx"
[ -n "$(ls -A "$VENDOR/dist" 2>/dev/null)" ] || { echo "roundtrip: vendor/dom-to-pptx not built — run scripts/setup-roundtrip.sh first" >&2; exit 1; }
mkdir -p "$OUT"
PPTX="$OUT/land-e-slides.pptx"; RT="$OUT/land-e-roundtrip.html"

echo "== html -> pptx (vendor/dom-to-pptx, nix chromium) =="
# LAND.E is a Marp deck: 1280×720 px = 13.333×7.5 in, slides are <section>
# elements (no .slide class — pass -s section). If the exporter warns about
# non-slide sections matching, narrow to -s 'section[id]'; the pptx must have
# exactly 20 slides (verify: our converter's slide count in the next stage).
nix shell nixpkgs#chromium --command bash -c \
  "cd '$VENDOR' && PUPPETEER_EXECUTABLE_PATH=\$(command -v chromium) \
   node bin/cli-exporter.js '$SRC' -o '$PPTX' -s section --title 'LAND.E' --width 13.333 --height 7.5"

echo "== pptx -> html (stevedore converter) =="
node "$ROOT/scripts/pptx-to-html.js" "$PPTX" -o "$RT"

echo "== compare =="
if node "$ROOT/scripts/compare-decks.mjs" "$SRC" "$RT"; then
  echo "ROUNDTRIP-OK"
else
  STATUS=$?
  echo "roundtrip: comparator FAILED (gates above) — artifacts kept in $OUT"
  exit $STATUS
fi
```

`chmod +x` it. Note: a comparator FAIL at this stage of the phase is information, not a crash — artifacts are kept for diagnosis (fidelity is plans 02-02..02-06's work).
