**Step 5 — export/html-to-pptx.sh (new; the export capability's standalone entry).** Create, `chmod +x`, full content:

```bash
#!/usr/bin/env bash
# export/html-to-pptx.sh — the export capability's standalone entry (phase 03,
# decision 1): html → pptx via the vendored dom-to-pptx fork, without running
# the whole round-trip. Wraps vendor/dom-to-pptx/bin/cli-exporter.js with
# stevedore defaults (the invocation roundtrip.sh used to bury as stage 1);
# nix chromium via PUPPETEER_EXECUTABLE_PATH — this wrapper self-wraps in
# `nix shell nixpkgs#chromium`, exactly like roundtrip.sh's stage 1 did.
# Usage: export/html-to-pptx.sh <input.html> -o <output.pptx> [options]
# Run scripts/setup-roundtrip.sh once first (builds vendor/dom-to-pptx/dist).
set -euo pipefail
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VENDOR="$ROOT/vendor/dom-to-pptx"

usage() {
  cat <<'EOF'
usage: export/html-to-pptx.sh <input.html> -o <output.pptx> [options]
  -s, --selector <css>   slide elements (default: section; .pp-slide for stevedore output)
      --title <text>     presentation title metadata (default: input basename)
      --width <in>       slide width in inches (default: 13.333)
      --height <in>      slide height in inches (default: 7.5)
  -h, --help             show this help
EOF
}

INPUT=""; OUT=""
SELECTOR="section"; TITLE=""; WIDTH="13.333"; HEIGHT="7.5"
while [ $# -gt 0 ]; do
  case "$1" in
    -o|-s|--selector|--title|--width|--height)
      [ $# -ge 2 ] || { echo "html-to-pptx: $1 requires a value" >&2; usage >&2; exit 2; } ;;
  esac
  case "$1" in
    -o) OUT="$2"; shift 2 ;;
    -s|--selector) SELECTOR="$2"; shift 2 ;;
    --title) TITLE="$2"; shift 2 ;;
    --width) WIDTH="$2"; shift 2 ;;
    --height) HEIGHT="$2"; shift 2 ;;
    -h|--help) usage; exit 0 ;;
    -*) echo "html-to-pptx: unknown option: $1" >&2; usage >&2; exit 2 ;;
    *) [ -z "$INPUT" ] || { echo "html-to-pptx: unexpected argument: $1" >&2; usage >&2; exit 2; }
       INPUT="$1"; shift ;;
  esac
done
[ -n "$INPUT" ] && [ -n "$OUT" ] || { usage >&2; exit 2; }
[ -f "$INPUT" ] || { echo "html-to-pptx: no such file: $INPUT" >&2; exit 2; }
[ -n "$(ls -A "$VENDOR/dist" 2>/dev/null)" ] || { echo "html-to-pptx: vendor/dom-to-pptx not built — run scripts/setup-roundtrip.sh first" >&2; exit 1; }
[ -z "$TITLE" ] && TITLE="$(basename "${INPUT%.*}")"

nix shell nixpkgs#chromium --command bash -c '
  cd "$1" && shift
  export PUPPETEER_EXECUTABLE_PATH="$(command -v chromium)"
  exec node bin/cli-exporter.js "$@"
' html-to-pptx "$VENDOR" "$INPUT" -o "$OUT" -s "$SELECTOR" --title "$TITLE" --width "$WIDTH" --height "$HEIGHT"
echo "html-to-pptx: wrote $OUT"
```

(The inner bash receives all variable data via argv — no quoting hazards; `command -v chromium` resolves INSIDE the nix shell. The vendored CLI's own defaults differ — `.slide`, 10×5.625 — this wrapper deliberately chooses stevedore-flavored defaults: `section`, 13.333×7.5, per roundtrip.sh's proven LAND.E invocation.)

```bash
#!/usr/bin/env bash
# scripts/roundtrip.sh — the whole round-trip from the combined repo (no /tmp
# tool cloning: tools are vendored; only outputs are scratch).
# Usage: scripts/roundtrip.sh <source.html> <outdir>
# Stages: html → pptx (export/html-to-pptx.sh → vendor/dom-to-pptx, nix
#         chromium) → html (extract/pptx-to-html.js — our converter)
#         → PNG exports of both sides (render/export-slides.mjs — the
#         deterministic exporter, plan 02-03)
#         → comparator (scripts/compare-decks.mjs).
# Run scripts/setup-roundtrip.sh once first.
set -euo pipefail
[ $# -eq 2 ] || { echo "usage: $0 <source.html> <outdir>" >&2; exit 2; }
SRC="$(realpath "$1")"; OUT="$(realpath -m "$2")"
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VENDOR="$ROOT/vendor/dom-to-pptx"
[ -n "$(ls -A "$VENDOR/dist" 2>/dev/null)" ] || { echo "roundtrip: vendor/dom-to-pptx not built — run scripts/setup-roundtrip.sh first" >&2; exit 1; }
mkdir -p "$OUT"
PPTX="$OUT/land-e-slides.pptx"; RT="$OUT/land-e-roundtrip.html"

echo "== html -> pptx (export/html-to-pptx.sh — vendor/dom-to-pptx, nix chromium) =="
# LAND.E is a Marp deck: 1280×720 px = 13.333×7.5 in, slides are <section>
# elements (no .slide class — pass -s section). If the exporter warns about
# non-slide sections matching, narrow to -s 'section[id]'; the pptx must have
# exactly 20 slides (verify: our converter's slide count in the next stage).
"$ROOT/export/html-to-pptx.sh" "$SRC" -o "$PPTX" -s section --title 'LAND.E' --width 13.333 --height 7.5

echo "== pptx -> html (extract/pptx-to-html.js — stevedore converter) =="
node "$ROOT/extract/pptx-to-html.js" "$PPTX" -o "$RT"

echo "== export PNGs (both sides, deterministic exporter) =="
nix shell nixpkgs#chromium --command bash -c \
  "STEVEDORE_CHROMIUM_PATH=\$(command -v chromium) node '$ROOT/render/export-slides.mjs' '$SRC' -o '$OUT/exports/reference'"
nix shell nixpkgs#chromium --command bash -c \
  "STEVEDORE_CHROMIUM_PATH=\$(command -v chromium) node '$ROOT/render/export-slides.mjs' '$RT' -o '$OUT/exports/roundtrip'"

echo "== compare =="
if node "$ROOT/scripts/compare-decks.mjs" "$SRC" "$RT"; then
  echo "ROUNDTRIP-OK"
else
  STATUS=$?
  echo "roundtrip: comparator FAILED (gates above) — artifacts kept in $OUT"
  exit $STATUS
fi
```
